async function connectDB() {
  const client = new MongoClient(MONGO_URI);
  await client.connect();
  db = client.db("juix");
  await db.collection("users").createIndex({ email: 1 }, { unique: true });
  await db.collection("sessions").createIndex({ createdAt: 1 }, { expireAfterSeconds: 60 * 60 * 24 * 7 });
  await db.collection("script_tokens").createIndex({ expiresAt: 1 }, { expireAfterSeconds: 120 });
  console.log("MongoDB connected");
}

function hashPw(pw) { return crypto.createHash("sha256").update(pw).digest("hex"); }

async function getUserRole(user) {
  if (!user) return "guest";
  if (SUPER_ADMIN_EMAIL && user.email === SUPER_ADMIN_EMAIL) return "super";
  if (user.role === "admin") return "admin";
  return "user";
}

async function requireAuth(req, res, next) {
  const token = req.headers["x-session"] || req.cookies.juix_session;
  if (!token) return res.status(401).json({ error: "not logged in" });
  const sess = await db.collection("sessions").findOne({ token });
  if (!sess) return res.status(401).json({ error: "session expired" });
  req.userId = sess.userId;
  req.user = await db.collection("users").findOne({ _id: sess.userId });
  if (!req.user) return res.status(401).json({ error: "user not found" });
  if (req.user.banned) {
    await db.collection("sessions").deleteOne({ token });
    return res.status(403).json({ error: "account banned" });
  }
  req.userRole = await getUserRole(req.user);
  next();
}

});

app.post("/api/key/validate", async (req, res) => {
  const { provider, key, hwid, username } = req.body;
  const p = await db.collection("providers").findOne({ id: provider });
  if (!p) return res.json({ valid: false, reason: "invalid provider" });
  if (p.enabled === false) return res.json({ valid: false, reason: "script disabled" });
  const k = await db.collection("keys").findOne({ key });
  if (!k) return res.json({ valid: false, reason: "invalid key" });
  if (k.provider !== provider) return res.json({ valid: false, reason: "key not for this provider" });
  if (k.banned) return res.json({ valid: false, reason: "key banned" });
  if (k.expires && Date.now() > new Date(k.expires).getTime()) return res.json({ valid: false, reason: "key expired" });
  const maxUses = k.maxUses || 1;
  const hwids = k.hwids || (k.hwid ? [k.hwid] : []);
  if (!hwids.includes(hwid)) {
    if (hwids.length >= maxUses) return res.json({ valid: false, reason: "max uses reached (" + maxUses + ")" });
    hwids.push(hwid);
    await db.collection("keys").updateOne({ key }, { $set: { hwids, hwid: hwids[0] } });
  }
  await db.collection("keyusers").updateOne({ hwid }, { $set: { hwid, username: username || "unknown", lastSeen: new Date() }, $setOnInsert: { firstSeen: new Date() }, $addToSet: { keys: key, providers: provider } }, { upsert: true });
  await db.collection("keys").updateOne({ key }, { $set: { lastUsed: new Date(), lastUser: username } });
  await db.collection("providers").updateOne({ id: provider }, { $inc: { hits: 1 } });
  const token = nanoid(32);
  await db.collection("script_tokens").insertOne({ token, providerId: provider, key, hwid, username, createdAt: new Date(), expiresAt: new Date(Date.now() + 60000), used: false });
  res.json({ valid: true, token });
});

function rc4(key, data) {
  const S = [];
  for (let i = 0; i < 256; i++) S[i] = i;
  const keyBytes = Buffer.from(key, "utf8");
  let j = 0;
  for (let i = 0; i < 256; i++) { j = (j + S[i] + keyBytes[i % keyBytes.length]) % 256; const tmp = S[i]; S[i] = S[j]; S[j] = tmp; }
  let i = 0; j = 0;
  const out = [];
  for (let k = 0; k < data.length; k++) { i = (i + 1) % 256; j = (j + S[i]) % 256; const tmp = S[i]; S[i] = S[j]; S[j] = tmp; out.push(data[k] ^ S[(S[i] + S[j]) % 256]); }
  return out;
}

function ironbrewCompile(script) {
  // Lightweight VM-style bytecode: PUSH_CONST / CONCAT / RETURN.
  // This is source reconstruction obfuscation, not native Lua bytecode.
  const bytes = Buffer.from(script, "utf8");
  const out = [];
  const KEY = crypto.randomInt(1, 256);

  // Per-build opcode IDs make the instruction stream less uniform.
  const opcodes = new Set();
  while (opcodes.size < 3) opcodes.add(crypto.randomInt(16, 256));
  const [PUSH, CONCAT, RETURN] = [...opcodes];

  // Header: magic, version, then this build's opcode map.
  out.push(0x7e, 0x02, PUSH, CONCAT, RETURN);

  // Split constants into varying-sized chunks. The VM joins them in order.
  let offset = 0;
  let pushed = 0;
  while (offset < bytes.length) {
    const chunkSize = crypto.randomInt(37, 89);
    const end = Math.min(bytes.length, offset + chunkSize);
    const chunk = bytes.subarray(offset, end);
    out.push(PUSH, (chunk.length >> 8) & 0xff, chunk.length & 0xff);
    for (const b of chunk) out.push(b);
    pushed++;
    if (pushed > 1) out.push(CONCAT);
    offset = end;
  }

  // Empty scripts still produce a valid empty constant.
  if (bytes.length === 0) out.push(PUSH, 0, 0);
  out.push(RETURN);

  const encrypted = out.map((b, i) => b ^ ((KEY + i * 13) % 256));
  return { bc: encrypted.join(","), key: KEY };
}

app.post("/api/script/proxy", async (req, res) => {
  const { token, hwid, providerId } = req.body;
  let providerIdToFetch = null;
  if (providerId) {
    const pDirect = await db.collection("providers").findOne({ id: providerId });
    if (pDirect && pDirect.keyless === true && pDirect.enabled !== false) {
      providerIdToFetch = pDirect.id;
      await db.collection("providers").updateOne({ id: providerId }, { $inc: { hits: 1 } });
    }
  }
  if (!providerIdToFetch) {
    if (!token || !hwid) return res.status(400).json({ error: "missing params" });
    const t = await db.collection("script_tokens").findOne({ token });
    if (!t) return res.status(404).json({ error: "invalid token" });
    if (t.used) return res.status(403).json({ error: "token used" });
    if (t.hwid !== hwid) return res.status(403).json({ error: "hwid mismatch" });
    if (Date.now() > new Date(t.expiresAt).getTime()) return res.status(410).json({ error: "token expired" });
    const consumed = await db.collection("script_tokens").updateOne(
      { token, used: false, hwid, expiresAt: { $gt: new Date() } },
      { $set: { used: true, usedAt: new Date() } }
    );
    if (consumed.modifiedCount !== 1) return res.status(403).json({ error: "token already consumed or expired" });
    providerIdToFetch = t.providerId;
  }
  const p = await db.collection("providers").findOne({ id: providerIdToFetch });
  if (!p) return res.status(404).json({ error: "provider not found" });
  if (p.enabled === false) return res.status(403).json({ error: "disabled" });
  const raw = (p.script || "").trim();
  let script;
  if (raw.startsWith("http://") || raw.startsWith("https://")) {
    const now = Date.now();
    const cached = scriptCache.get(raw);
    if (cached && now - cached.time < 5 * 60 * 1000) script = cached.script;
    else {
      try {
        const r = await fetchFn(raw, { headers: { "User-Agent": "JUIX-Proxy/1.0" } });
        if (!r.ok) return res.status(500).json({ error: "remote fetch failed" });
        script = await r.text();
        scriptCache.set(raw, { script, time: now });
      } catch (e) { return res.status(500).json({ error: e.message }); }
    }
  } else script = raw;

  const compiled = ironbrewCompile(script);

  const RC4_KEY = crypto.randomBytes(24).toString("hex");
  const utf8Bytes = Buffer.from(compiled.bc, "utf8");
  const rc4Out = rc4(RC4_KEY, utf8Bytes);
  const XOR_KEY = crypto.randomInt(1, 256);
  const enc = [];
  for (let i = 0; i < rc4Out.length; i++) enc.push(rc4Out[i] ^ ((XOR_KEY + i * 7) % 256));

  res.setHeader("Cache-Control", "no-store");
  res.json({ ok: true, enc: enc.join(","), ibk: compiled.key, rk: RC4_KEY, xk: XOR_KEY });
});

local function _is_native(fn)
    if type(fn) ~= "function" then return false end

            if not _EXEC_SAFE then return false end

    end
    return true

local function _run(enc, ibk, rk, xk)
    if not _check_hook() then _genv.JUIX_LOCK = nil; return end
    task.spawn(function()
        while task.wait(30) do
            pcall(function()
                if not _check_hook() then return end
                _REF.request({
                    Url=_api.."/api/heartbeat",
                    Method="POST",
                    Headers={["Content-Type"]="application/json"},
                    Body=_svc:JSONEncode({hwid=_hwid}),
                })
            end)
        end
    end)

    local bytes = {}
    for n in enc:gmatch("([^,]+)") do
        local num = tonumber(n)
        if num then bytes[#bytes+1] = num end
    end

    local XOX_KEY = tonumber(xk) or 218
    local rc4In = {}
    for i = 1, #bytes do
        local b = _REF.bxor(bytes[i], (XOX_KEY + (i - 1) * 7) % 256)
        if b < 0 or b > 255 then b = b % 256 end
        rc4In[i] = b
    end

    local RC4_KEY = tostring(rk or "")
    if RC4_KEY == "" then _genv.JUIX_LOCK = nil; return end
    local S = {}
    for i = 0, 255 do S[i] = i end
    local j = 0
    for i = 0, 255 do
        j = (j + S[i] + _REF.byte(RC4_KEY, (i % #RC4_KEY) + 1)) % 256
        S[i], S[j] = S[j], S[i]
    end
    local ii, jj = 0, 0
    local chars = {}
    for k = 1, #rc4In do
        ii = (ii + 1) % 256
        jj = (jj + S[ii]) % 256
        S[ii], S[jj] = S[jj], S[ii]
        local K = S[(S[ii] + S[jj]) % 256]
        local b = _REF.bxor(rc4In[k], K)
        if b < 0 or b > 255 then b = b % 256 end
        chars[k] = _REF.char(b)
    end
    local bcStr = _REF.concat(chars)

    local bc = {}
    for n in bcStr:gmatch("([^,]+)") do
        local num = tonumber(n)
        if num then bc[#bc+1] = num end
    end

    local KEY = tonumber(ibk) or 0x5A
    local decrypted = {}
    for i = 1, #bc do
        local b = _REF.bxor(bc[i], (KEY + (i - 1) * 13) % 256)
        if b < 0 or b > 255 then b = b % 256 end
        decrypted[i] = b
    end

    -- Execute the compact VM instruction stream.
    -- PUSH reads a length-prefixed constant; CONCAT joins the top two values;
    -- RETURN yields the reconstructed source for the compatibility loader.
    local stack = {}
    local pc = 1
    local finalSrc = nil
    if decrypted[pc] ~= 0x7e or decrypted[pc + 1] ~= 0x02 then
        _genv.JUIX_LOCK = nil
        return
    end
    local PUSH = decrypted[pc + 2]
    local CONCAT = decrypted[pc + 3]
    local RETURN = decrypted[pc + 4]
    if not PUSH or not CONCAT or not RETURN
        or PUSH == CONCAT or PUSH == RETURN or CONCAT == RETURN then
        _genv.JUIX_LOCK = nil
        return
    end
    pc = pc + 5

    while pc <= #decrypted do
        local op = decrypted[pc]
        pc = pc + 1
        if op == PUSH then
            if pc + 1 > #decrypted then _genv.JUIX_LOCK = nil; return end
            local len = decrypted[pc] * 256 + decrypted[pc + 1]
            pc = pc + 2
            if len < 0 or pc + len - 1 > #decrypted then
                _genv.JUIX_LOCK = nil
                return
            end
            local chars = {}
            for i = 1, len do
                chars[i] = _REF.char(decrypted[pc])
                pc = pc + 1
            end
            stack[#stack + 1] = _REF.concat(chars)
        elseif op == CONCAT then
            if #stack < 2 then _genv.JUIX_LOCK = nil; return end
            local a = stack[#stack - 1]
            local b = stack[#stack]
            stack[#stack - 1] = a .. b
            stack[#stack] = nil
        elseif op == RETURN then
            if #stack ~= 1 then _genv.JUIX_LOCK = nil; return end
            finalSrc = stack[1]
            break
        else
            _genv.JUIX_LOCK = nil
            return
        end
    end

    if type(finalSrc) ~= "string" then
        _genv.JUIX_LOCK = nil
        return
    end

    if not _check_hook() then _genv.JUIX_LOCK = nil; return end
    local fn,err = _REF.loadstring(finalSrc)
    if not fn then _genv.JUIX_LOCK = nil; return end
    local ok,re = pcall(fn)
    if not ok then end
end

local function dragify(frame,handle,onMove)
    local dragging,dragStart,startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then

            input.Changed:Connect(function()
                if input.UserInputState==Enum.UserInputState.End then dragging=false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch) then
            local d=input.Position-dragStart
